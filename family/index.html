      ☁️ Supabase 연결 (가족 실시간 동기화)

      프로젝트 URL과 anon public key를 넣으면, 로그인한 가족 계정끼리 사진·일정·위치가 실시간으로 맞춰져요.

      처음 한 번: Supabase에 표 만들기 (SQL 복사) 
      -- Supabase → SQL Editor 에 붙여넣고 RUN
create table if not exists public.family_dashboard_state (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz not null default now()
);

alter table public.family_dashboard_state enable row level security;

drop policy if exists "family read"   on public.family_dashboard_state;
drop policy if exists "family insert" on public.family_dashboard_state;
drop policy if exists "family update" on public.family_dashboard_state;

-- 로그인한 가족 계정만 읽고 쓸 수 있어요
create policy "family read"   on public.family_dashboard_state
  for select to authenticated using (true);
create policy "family insert" on public.family_dashboard_state
  for insert to authenticated with check (true);
create policy "family update" on public.family_dashboard_state
  for update to authenticated using (true) with check (true);

-- 실시간 구독 켜기
alter publication supabase_realtime add table public.family_dashboard_state;

      Authentication → Users 에서 가족 이메일 계정을 만들고(비밀번호 직접 지정), Email 로그인을 켜 두면 돼요. 같은 표에 다른 사람이 들어오지 못하도록 RLS 정책은 '로그인한 사용자만'으로 잠가 둡니다.

      💾 백업 / 복원

      지금 화면의 내용을 JSON 파일로 내려받거나, 다른 폰에서 그대로 불러올 수 있어요.

      🔒 사진과 개인정보

      올린 사진은 폰 안에서 먼저 줄여 저장해요 — 프로필 256×256, 위치 아이콘 128×128, 가족 사진 액자는 한 장당 최대 800px(WebP 0.80, 최대 12장). 공개 주소의 페이지 코드에는 사진·일정·위치가 들어가지 않고, 기기 저장소(또는 로그인한 Supabase)에만 남아요.
